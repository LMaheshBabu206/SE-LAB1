# Lab 5-47: Static Code Analysis – Domain & SSL Expiry Alert System

# 1. Create and activate virtual environment (Windows PowerShell)
py -m venv .venv
.venv\Scripts\Activate.ps1
python -m pip install -r requirements.txt

# 2. Run the initial demonstration (offline demo; RDAP helper is not called)
python domain_ssl_monitor_before.py

# 3. Run static analysis on the initial, intentionally flawed file
python -m pylint domain_ssl_monitor_before.py > reports/pylint_report_before.txt
python -m bandit -r domain_ssl_monitor_before.py > reports/bandit_report_before.txt
python -m flake8 domain_ssl_monitor_before.py > reports/flake8_report_before.txt

# 4. Run the corrected version and tests
python domain_ssl_monitor.py
python -m pytest -v

# 5. Run static analysis on the corrected version
python -m pylint domain_ssl_monitor.py > reports/pylint_report_after.txt
python -m bandit -r domain_ssl_monitor.py > reports/bandit_report_after.txt
python -m flake8 domain_ssl_monitor.py > reports/flake8_report_after.txt
